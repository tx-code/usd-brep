cmake_minimum_required(VERSION 3.22)
project(planar_transfer_contracts LANGUAGES CXX)
enable_testing()
set(SMLIB_BUILD_VALIDATION OFF CACHE BOOL "Build the wider SDK validation targets")
set(SMLIB_BUILD_DEV OFF CACHE BOOL "" FORCE)
set(SMLIB_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../.. sdk)
# Fault injection is confined to this dedicated build, never the normal SDK.
target_compile_definitions(SMLib PRIVATE SM_PLANAR_TRANSFER_TESTING)
add_executable(planar_transfer_contracts contracts.cpp)
target_compile_definitions(planar_transfer_contracts PRIVATE SM_PLANAR_TRANSFER_TESTING)
target_compile_features(planar_transfer_contracts PRIVATE cxx_std_20)
target_link_libraries(planar_transfer_contracts PRIVATE SM_API)
set_target_properties(planar_transfer_contracts PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_test(NAME planar_transfer_contracts COMMAND planar_transfer_contracts)
add_test(NAME planar_transfer_degenerate COMMAND planar_transfer_contracts degenerate)
set_tests_properties(planar_transfer_contracts PROPERTIES TIMEOUT 120)
set_tests_properties(planar_transfer_degenerate PROPERTIES TIMEOUT 30)
if(WIN32)
    add_test(NAME planar_transfer_retention COMMAND planar_transfer_contracts resources CONFIGURATIONS Debug)
    set_tests_properties(planar_transfer_retention PROPERTIES TIMEOUT 120)
endif()

option(PLANAR_TRANSFER_BUILD_PYTHON "Build bindings for existing public regression tests" OFF)
if(PLANAR_TRANSFER_BUILD_PYTHON)
    get_filename_component(repo_root "../../../.." ABSOLUTE)
    find_package(Python3 3.12 EXACT REQUIRED COMPONENTS Interpreter Development.Module)
    set(PYBIND11_FINDPYTHON ON)
    if(EXISTS "${repo_root}/_build/target-deps/pybind11/CMakeLists.txt")
        add_subdirectory("${repo_root}/_build/target-deps/pybind11" pybind11 EXCLUDE_FROM_ALL)
    else()
        find_package(pybind11 CONFIG REQUIRED)
    endif()
    file(GLOB binding_sources CONFIGURE_DEPENDS "${repo_root}/source/SmPyLib/src/*.cpp")
    pybind11_add_module(_omni_solid MODULE NO_EXTRAS ${binding_sources})
    target_compile_features(_omni_solid PRIVATE cxx_std_20)
    target_link_libraries(_omni_solid PRIVATE SM_API)
    set_target_properties(_omni_solid PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
