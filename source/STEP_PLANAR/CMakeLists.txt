cmake_minimum_required(VERSION 3.20)
project(STEP_PLANAR LANGUAGES CXX)
enable_testing()
add_subdirectory(../STEP_SEMANTICS source-semantics)
add_library(STEP_PLANAR STATIC src/StepExact.cpp src/StepPlanar.cpp src/StepPlanarRules.cpp
    src/StepPlanarGeometry.cpp src/StepPlanarJson.cpp)
target_include_directories(STEP_PLANAR PUBLIC inc PRIVATE src)
target_compile_features(STEP_PLANAR PUBLIC cxx_std_20)
target_link_libraries(STEP_PLANAR PUBLIC STEP_SEMANTICS)
add_executable(step_planar_inspect app/step_planar.cpp)
target_link_libraries(step_planar_inspect PRIVATE STEP_PLANAR)
option(STEP_PLANAR_BUILD_TESTS "Build source planar rule and region contract tests" ON)
if(STEP_PLANAR_BUILD_TESTS)
    add_executable(step_planar_tests tests/StepPlanarTests.cpp)
    target_link_libraries(step_planar_tests PRIVATE STEP_PLANAR)
    add_test(NAME step_planar_contracts COMMAND step_planar_tests
        "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/step_import_validation/corpus/ap203e2-planar/v1/cases")
endif()
foreach(target STEP_PLANAR step_planar_inspect step_planar_tests)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
    endif()
endforeach()
