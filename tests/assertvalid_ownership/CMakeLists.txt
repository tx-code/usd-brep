cmake_minimum_required(VERSION 3.22)
project(AssertValidOwnershipRegression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
get_filename_component(SMLIB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SMLIB_LIBRARY_DIR "" CACHE PATH "Directory containing SMLib and SM_API libraries")
set(SMLIB_RUNTIME_DIR "${SMLIB_LIBRARY_DIR}" CACHE PATH "Directory containing matching runtime libraries")
add_executable(assertvalid_ownership ownership.cpp)
target_include_directories(assertvalid_ownership PRIVATE "${SMLIB_ROOT}/source/SMLib/inc" "${SMLIB_ROOT}/source/SM_API/inc")
target_link_directories(assertvalid_ownership PRIVATE "${SMLIB_LIBRARY_DIR}")
target_link_libraries(assertvalid_ownership PRIVATE SM_API SMLib)
if(WIN32)
    target_compile_options(assertvalid_ownership PRIVATE /utf-8)
    target_compile_definitions(assertvalid_ownership PRIVATE NOMINMAX UNICODE _UNICODE SM_GRAPHICS_CALLBACKS=1)
endif()
enable_testing()
add_test(NAME seam-classification-aliases COMMAND assertvalid_ownership --aliases)
set(ownership_tests seam-classification-aliases)
foreach(case owned borrowed external context empty self-uv borrowed-surface borrowed-children transferred borrowed-intervals)
    add_test(NAME classification-copy-${case} COMMAND assertvalid_ownership --copy-${case})
    list(APPEND ownership_tests classification-copy-${case})
endforeach()
if(WIN32)
    add_test(NAME assertvalid-allocation-balance COMMAND assertvalid_ownership --memory)
    add_test(NAME classification-copy-allocation-balance COMMAND assertvalid_ownership --copy-memory)
    list(APPEND ownership_tests assertvalid-allocation-balance classification-copy-allocation-balance)
    set_tests_properties(${ownership_tests} PROPERTIES
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${SMLIB_RUNTIME_DIR}")
endif()
