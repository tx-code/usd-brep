cmake_minimum_required(VERSION 3.22)
project(CurveValidationRegression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
get_filename_component(SMLIB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SMLIB_LIBRARY_DIR "" CACHE PATH "Directory containing SMLib libraries")
set(SMLIB_RUNTIME_DIR "${SMLIB_LIBRARY_DIR}" CACHE PATH "Directory containing matching runtime libraries")
add_executable(curve_validation validation.cpp)
target_include_directories(curve_validation PRIVATE "${SMLIB_ROOT}/source/SMLib/inc")
target_link_directories(curve_validation PRIVATE "${SMLIB_LIBRARY_DIR}")
target_link_libraries(curve_validation PRIVATE SMLib)
if(WIN32)
    target_compile_options(curve_validation PRIVATE /utf-8)
    target_compile_definitions(curve_validation PRIVATE NOMINMAX UNICODE _UNICODE SM_GRAPHICS_CALLBACKS=1 NL_DEBUG_CODE)
endif()
enable_testing()
add_test(NAME curve-validation COMMAND curve_validation)
if(WIN32)
    set_tests_properties(curve-validation PROPERTIES
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${SMLIB_RUNTIME_DIR}")
endif()
