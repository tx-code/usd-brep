cmake_minimum_required(VERSION 3.22)
project(PlanarRegionRegression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
get_filename_component(SMLIB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SMLIB_LIBRARY_DIR "" CACHE PATH "Directory containing SMLib and SM_API libraries")
set(SMLIB_RUNTIME_DIR "${SMLIB_LIBRARY_DIR}" CACHE PATH "Directory containing runtime libraries")
add_executable(planar_regions regions.cpp)
target_include_directories(planar_regions PRIVATE "${SMLIB_ROOT}/source/SMLib/inc" "${SMLIB_ROOT}/source/SM_API/inc")
target_link_directories(planar_regions PRIVATE "${SMLIB_LIBRARY_DIR}")
target_link_libraries(planar_regions PRIVATE SM_API SMLib)
if(WIN32)
    target_compile_options(planar_regions PRIVATE /utf-8)
    target_compile_definitions(planar_regions PRIVATE NOMINMAX UNICODE _UNICODE SM_GRAPHICS_CALLBACKS=1)
endif()
enable_testing()
add_test(NAME planar-regions COMMAND planar_regions)
if(WIN32)
    set_tests_properties(planar-regions PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${SMLIB_RUNTIME_DIR}")
endif()
